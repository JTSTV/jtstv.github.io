async function copyCode(text){
  // 优先现代Clipboard API
  if(navigator.clipboard && window.isSecureContext){
    try{
      await navigator.clipboard.writeText(text);
      return true;
    }catch(e){
      // 失败走降级方案
    }
  }
  // 降级：传统textarea同步复制，适配鸿蒙安卓
  const ta = document.createElement('textarea');
  ta.value = text;
  ta.readOnly = true;
  ta.style.position = 'fixed';
  ta.style.left = '-9999px';
  document.body.appendChild(ta);
  ta.select();
  const ok = document.execCommand('copy');
  document.body.removeChild(ta);
  return ok;
}

<pre><code class="language-yaml">name: Auto-Post
on:
  push:
    branches: [ main ]

jobs:
  convert_raw_post:
    runs-on: ubuntu-latest
    steps:
      - name: 拉取代码
        uses: actions/checkout@v4
        with:
          fetch-depth: 0
          token: ${{ secrets.GITHUB_TOKEN }}

      - name: 设置时区
        run: |
          sudo ln -sf /usr/share/zoneinfo/Asia/Shanghai /etc/localtime
          echo "Asia/Shanghai" | sudo tee /etc/timezone

      - name: 判断跳过标记
        id: skip_check
        run: |
          MSG="${{ github.event.head_commit.message }}"
          if [[ "${MSG}" == *"[SKIP_AUTO_TITLE]"* ]];then
            echo "skip=true" >> $GITHUB_OUTPUT
          else
            echo "skip=false" >> $GITHUB_OUTPUT
          fi

      - name: 遍历转换，规避子shell
        if: steps.skip_check.outputs.skip == 'false'
        run: |
          POST_DIR="_posts"
          TODAY=$(date +%Y-%m-%d)
          mkdir -p "${POST_DIR}"
          CHANGED_FILES=()

          echo "====开始扫描===="
          while IFS= read -r -d '' filepath; do
            filename=$(basename "${filepath}")
            echo "扫描文件: ${filename}"

            [[ "${filename}" == *.md ]] && continue
            echo "${filename}" | grep -E '^[0-9]{4}-[0-9]{2}-[0-9]{2}-' >/dev/null && continue

            TMP_RAW=$(mktemp /tmp/post.XXXXXX)
            ESC_TITLE=$(printf "%s" "${filename}" | sed 's/"/\\"/g')
            OUT_MD="${POST_DIR}/${TODAY}-${filename}.md"

            {
              echo '---'
              echo "title: \"${ESC_TITLE}\""
              echo "date: ${TODAY}"
              echo '---'
              cat "${filepath}"
            } > "${TMP_RAW}"

            rm -f "${filepath}"
            mv "${TMP_RAW}" "${OUT_MD}"
            CHANGED_FILES+=("${OUT_MD}")
            echo "✅转换完成 → ${OUT_MD}"

          done < <(find "${POST_DIR}" -maxdepth 1 -type f -print0)

          echo "共转换数量：${#CHANGED_FILES[@]}"
          if [[ ${#CHANGED_FILES[@]} -eq 0 ]];then
            echo "无需要转换文件，退出"
            exit 0
          fi

          git config --global user.name "KBot-Auto"
          git config --global user.email "bot@example.com"
          git add "${POST_DIR}/"
          git commit -m "[SKIP_AUTO_TITLE] auto:转换无后缀文章"
          git push origin main
